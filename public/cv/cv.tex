% ========== Selected Research Projects ==========
\section{\textbf{Selected Research Projects}}
\vspace{-0.4mm}
\resumeSubHeadingListStart

\resumeProject
  {SIGNPOST-Bench: Benchmarking Text--Vision Conflict Resolution in MLLMs}
  {First Author \& Project Lead (In preparation for AAAI 2027)}
  {2025 -- Present}
  {{}}
\resumeItemListStart
  \item Proposed a counterfactual benchmark that isolates text--vision conflict in multimodal reasoning, comprising 5,111 image groups and 25,000+ variants across four geo-localization datasets, with an automated pipeline for adversarial text generation and scene-coherent image editing.
  \item Evaluated 20 MLLMs from seven providers and found that adversarial text increases median prediction error by 4.8$\times$; introduced the \textbf{Multimodal Conflict Robustness Score (MCRS)}, showing that conflict-resolution capability is orthogonal to clean-input accuracy and does not scale with model size.
\resumeItemListEnd

\resumeProject
  {MACRO-City Engine}
  {First Author (In preparation for \textit{Nature Cities})}
  {2025 -- Present}
  {{}}
\resumeItemListStart
  \item Investigating ``fragile booms'' by quantifying the temporal mismatch between rapid urban activity growth and slow infrastructure accumulation across 302 global cities, using a satellite-based stock-flow diagnostic built from 12 geospatial data sources and validated via COVID-19 as an exogenous shock.
\resumeItemListEnd

\resumeProject
  {GLARE (Geolocation Leakage Attention \& Reasoning Engine)}
  {Project Lead \& Core Researcher}
  {2025 -- Present}
  {{}}
\resumeItemListStart
  \item Designed an MLLM research framework that triangulates white-box cross-attention grounding, verbalized clue analysis, and human mouse-attention proxies to study geographic privacy leakage in street-view imagery.
  \item Built an end-to-end pipeline for attention extraction, calibrated risk scoring, and bias analysis with statistical rigor (bootstrap CI, FDR correction, effect sizes), formulating four research questions covering internal--output alignment, model--human attention gap, predictive validity, and systematic bias.
\resumeItemListEnd
